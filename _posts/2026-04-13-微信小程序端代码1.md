## 1. app.js（小程序入口文件，初始化全局配置）

```javascript
// app.js
App({
  // 全局数据（存储用户token、用户名等）
  globalData: {
    userInfo: null,
    token: wx.getStorageSync('token') || '', // 从本地缓存获取token
    baseUrl: 'http://127.0.0.1:8000/api' // 后端接口基础地址（本地调试用）
  },

  // 小程序初始化完成时执行
  onLaunch() {
    // 检查token是否有效，无效则跳转至登录页
    if (this.globalData.token) {
      this.checkTokenValid()
    } else {
      wx.navigateTo({ url: '/pages/login/login' })
    }
  },

  // 检查token有效性
  checkTokenValid() {
    const that = this
    wx.request({
      url: `${this.globalData.baseUrl}/user/checkToken`,
      method: 'GET',
      header: {
        'Authorization': `Bearer ${this.globalData.token}`
      },
      success(res) {
        if (res.data.code !== 200) {
          // token无效或过期，清除缓存并跳转登录页
          wx.removeStorageSync('token')
          that.globalData.token = ''
          wx.navigateTo({ url: '/pages/login/login' })
        }
      },
      fail() {
        wx.showToast({ title: '网络异常，请检查连接&#39;, icon: &#39;none&#39; })
      }
    })
  }
})
```

## 4. pages/login/login.js（登录页面逻辑，含注册功能）

```javascript
// pages/login/login.js
const app = getApp()
Page({
  data: {
    isLogin: true, // true：登录表单，false：注册表单
    loginForm: {
      username: '',
      password: ''
    },
    registerForm: {
      username: '',
      password: '',
      confirmPwd: ''
    },
    isLoading: false // 加载状态
  },

  // 切换登录/注册表单
  toggleForm() {
    this.setData({ isLogin: !this.data.isLogin })
  },

  // 输入框内容变化（登录表单）
  onLoginInput(e) {
    const { name } = e.currentTarget.dataset
    this.setData({
      [`loginForm.${name}`]: e.detail.value
    })
  },

  // 输入框内容变化（注册表单）
  onRegisterInput(e) {
    const { name } = e.currentTarget.dataset
    this.setData({
      [`registerForm.${name}`]: e.detail.value
    })
  },

  // 登录提交
  async login() {
    const { username, password } = this.data.loginForm
    // 表单验证
    if (!username.trim() || !password.trim()) {
      wx.showToast({ title: '用户名和密码不能为空', icon: 'none' })
      return
    }
    if (username.length < 3 || username.length > 50) {
      wx.showToast({ title: '用户名长度需在3-50位之间', icon: 'none' })
      return
    }
    if (password.length < 6 || password.length > 20) {
      wx.showToast({ title: '密码长度需在6-20位之间', icon: 'none' })
      return
    }

    this.setData({ isLoading: true })
    try {
      // 调用后端登录接口
      const res = await wx.request({
        url: `${app.globalData.baseUrl}/user/login`,
        method: 'POST',
        header: { 'Content-Type': 'application/json' },
        data: { username, password }
      })

      if (res.data.code === 200) {
        // 登录成功，存储token并跳转至对话页
        const { token, username } = res.data.data
        wx.setStorageSync('token', token)
        app.globalData.token = token
        app.globalData.userInfo = { username }
        wx.navigateTo({ url: '/pages/chat/chat' })
        wx.showToast({ title: '登录成功' })
      } else {
        wx.showToast({ title: res.data.msg || '登录失败', icon: 'none' })
      }
    } catch (err) {
      wx.showToast({ title: '网络异常，请重试', icon: 'none' })
      console.error('登录失败：', err)
    } finally {
      this.setData({ isLoading: false })
    }
  },

  // 注册提交
  async register() {
    const { username, password, confirmPwd } = this.data.registerForm
    // 表单验证
    if (!username.trim() || !password.trim() || !confirmPwd.trim()) {
      wx.showToast({ title: '所有字段不能为空', icon: 'none' })
      return
    }
    if (username.length < 3 || username.length > 50) {
      wx.showToast({ title: '用户名长度需在3-50位之间', icon: 'none' })
      return
    }
    if (password.length < 6 || password.length > 20) {
      wx.showToast({ title: '密码长度需在6-20位之间', icon: 'none' })
      return
    }
    if (password !== confirmPwd) {
      wx.showToast({ title: '两次输入的密码不一致', icon: 'none' })
      return
    }

    this.setData({ isLoading: true })
    try {
      // 调用后端注册接口
      const res = await wx.request({
        url: `${app.globalData.baseUrl}/user/register`,
        method: 'POST',
        header: { 'Content-Type': 'application/json' },
        data: { username, password }
      })

      if (res.data.code === 200) {
        wx.showToast({ title: res.data.msg || '注册成功，请登录' })
        // 切换至登录表单
        this.setData({ isLogin: true })
      } else {
        wx.showToast({ title: res.data.msg || '注册失败', icon: 'none' })
      }
    } catch (err) {
      wx.showToast({ title: '网络异常，请重试', icon: 'none' })
      console.error('注册失败：', err)
    } finally {
      this.setData({ isLoading: false })
    }
  }
})
```

## 8. pages/chat/chat.js（AI流式对话页面逻辑）

```javascript
// pages/chat/chat.js
const app = getApp()
Page({
  data: {
    message: '', // 用户输入的消息
    chatList: [], // 对话历史列表
    isLoading: false, // 加载状态
    isStreaming: false // 流式输出状态
  },

  // 页面加载时获取历史对话（可选，需后端支持）
  onLoad() {
    this.getChatHistory()
  },

  // 获取对话历史
  async getChatHistory() {
    try {
      const res = await wx.request({
        url: `${app.globalData.baseUrl}/chat/history`,
        method: 'GET',
        header: {
          'Authorization': `Bearer ${app.globalData.token}`
        }
      })
      if (res.data.code === 200) {
        this.setData({ chatList: res.data.data })
      }
    } catch (err) {
      console.error('获取对话历史失败：', err)
    }
  },

  // 输入框内容变化
  onInput(e) {
    this.setData({ message: e.detail.value })
  },

  // 发送消息（流式对话）
  sendMessage() {
    const { message, isLoading, isStreaming } = this.data
    const { token } = app.globalData

    // 验证
    if (!message.trim()) {
      wx.showToast({ title: '请输入提问内容', icon: 'none' })
      return
    }
    if (isLoading || isStreaming) {
      wx.showToast({ title: '请等待当前请求完成', icon: 'none' })
      return
    }

    // 添加用户消息到对话列表
    const newChatList = [
      ...this.data.chatList,
      { role: 'user', content: message.trim() }
    ]
    this.setData({
      chatList: newChatList,
      message: '',
      isStreaming: true,
      // 先添加一个空的助手消息占位
      chatList: [...newChatList, { role: 'assistant', content: '' }]
    })

    // 调用流式对话接口
    this.streamChat(message.trim())
  },

  // 流式对话核心方法
  async streamChat(message) {
    const { token } = app.globalData
    const that = this

    // 小程序流式请求（使用wx.request的responseType: 'arraybuffer'）
    const task = wx.request({
      url: `${app.globalData.baseUrl}/chat`,
      method: 'POST',
      header: {
        'Authorization': `Bearer ${token}`,
        'Content-Type': 'application/json'
      },
      data: {
        message,
        history: this.data.chatList.slice(0, -1) // 排除最后一个空占位
      },
      responseType: 'arraybuffer',
      success(res) {
        // 流式响应处理（解析SSE格式）
        const decoder = new TextDecoder('utf-8')
        const data = decoder.decode(res.data)
        const lines = data.split('\n\n')

        lines.forEach(line => {
          if (line.startsWith('data: ')) {
            try {
              const jsonStr = line.slice(6).trim()
              if (jsonStr) {
                const jsonData = JSON.parse(jsonStr)
                const content = jsonData.content || ''
                if (content) {
                  // 更新助手消息内容
                  const chatList = that.data.chatList
                  chatList[chatList.length - 1].content += content
                  that.setData({ chatList })
                }
              }
            } catch (err) {
              console.error('解析流式响应失败：', err)
            }
          }
        })
      },
      fail(err) {
        wx.showToast({ title: '对话失败，请重试', icon: 'none' })
        console.error('流式对话失败：', err)
        // 移除空占位
        const chatList = that.data.chatList.slice(0, -1)
        that.setData({ chatList })
      },
      complete() {
        that.setData({ isStreaming: false, isLoading: false })
      }
    })

    // 取消请求（可选）
    this.setData({ cancelTask: task })
  },

  // 取消当前请求
  cancelRequest() {
    const { cancelTask, isStreaming } = this.data
    if (isStreaming && cancelTask) {
      cancelTask.abort()
      this.setData({ isStreaming: false, isLoading: false })
    }
  },

  // 清空对话历史
  clearChat() {
    this.setData({ chatList: [] })
    // 可选：调用后端接口清空历史
    wx.request({
      url: `${app.globalData.baseUrl}/chat/clear`,
      method: 'POST',
      header: {
        'Authorization': `Bearer ${app.globalData.token}`
      }
    })
  }
})
```

## 12. pages/knowledge/knowledge.js（知识库页面逻辑）

```javascript
// pages/knowledge/knowledge.js
const app = getApp()
Page({
  data: {
    question: '', // 查询问题
    answer: '', // 查询答案
    fileList: [], // 已上传文件列表
    isLoading: false, // 加载状态
    uploadProgress: 0 // 上传进度
  },

  // 页面加载时获取已上传文件
  onLoad() {
    this.getUploadedFiles()
  },

  // 获取已上传的知识库文件
  async getUploadedFiles() {
    try {
      const res = await wx.request({
        url: `${app.globalData.baseUrl}/knowledge/files`,
        method: 'GET',
        header: {
          'Authorization': `Bearer ${app.globalData.token}`
        }
      })
      if (res.data.code === 200) {
        this.setData({ fileList: res.data.data })
      }
    } catch (err) {
      wx.showToast({ title: '获取文件列表失败', icon: 'none' })
      console.error('获取文件列表失败：', err)
    }
  },

  // 输入框内容变化
  onInput(e) {
    this.setData({ question: e.detail.value })
  },

  // 上传知识库文件
  chooseFile() {
    const that = this
    // 选择文件（支持PDF、TXT、DOCX）
    wx.chooseMessageFile({
      count: 1,
      type: 'file',
      extension: ['pdf', 'txt', 'docx'],
      success(res) {
        const tempFile = res.tempFiles[0]
        // 验证文件格式
        const fileExt = tempFile.name.split('.').pop().toLowerCase()
        if (!['pdf', 'txt', 'docx'].includes(fileExt)) {
          wx.showToast({ title: '仅支持PDF、TXT、DOCX格式', icon: 'none' })
          return
        }

        // 上传文件到后端
        wx.uploadFile({
          url: `${app.globalData.baseUrl}/knowledge/upload`,
          filePath: tempFile.path,
          name: 'file',
          header: {
            'Authorization': `Bearer ${app.globalData.token}`
          },
          formData: { filename: tempFile.name },
          progress(progress) {
            // 更新上传进度
            that.setData({ uploadProgress: Math.floor(progress.progress * 100) })
          },
          success(uploadRes) {
            const data = JSON.parse(uploadRes.data)
            if (data.code === 200) {
              wx.showToast({ title: '文件上传成功' })
              // 刷新文件列表
              that.getUploadedFiles()
            } else {
              wx.showToast({ title: data.msg || '上传失败', icon: 'none' })
            }
          },
          fail(err) {
            wx.showToast({ title: '上传失败，请重试', icon: 'none' })
            console.error('文件上传失败：', err)
          },
          complete() {
            that.setData({ uploadProgress: 0 })
          }
        })
      }
    })
  },

  // 知识库查询
  async queryKnowledge() {
    const { question, isLoading } = this.data
    if (!question.trim()) {
      wx.showToast({ title: '请输入查询问题', icon: 'none' })
      return
    }
    if (isLoading) {
      wx.showToast({ title: '请等待查询完成', icon: 'none' })
      return
    }

    this.setData({ isLoading: true, answer: '' })
    try {
      const res = await wx.request({
        url: `${app.globalData.baseUrl}/knowledge/query`,
        method: 'POST',
        header: {
          'Authorization': `Bearer ${app.globalData.token}`,
          'Content-Type': 'application/json'
        },
        data: { question: question.trim() }
      })

      if (res.data.code === 200) {
        this.setData({ answer: res.data.data.answer })
      } else {
        this.setData({ answer: res.data.msg || '查询失败' })
      }
    } catch (err) {
      this.setData({ answer: '网络异常，请重试' })
      console.error('知识库查询失败：', err)
    } finally {
      this.setData({ isLoading: false })
    }
  },

  // 删除已上传文件
  async deleteFile(e) {
    const { filename } = e.currentTarget.dataset
    wx.showModal({
      title: '确认删除',
      content: `确定要删除文件【${filename}】吗？`,
      confirmText: '删除',
      cancelText: '取消',
      async success(res) {
        if (res.confirm) {
          try {
            const deleteRes = await wx.request({
              url: `${app.globalData.baseUrl}/knowledge/delete`,
              method: 'POST',
              header: {
                'Authorization': `Bearer ${app.globalData.token}`
              },
              data: { filename }
            })
            if (deleteRes.data.code === 200) {
              wx.showToast({ title: '删除成功' })
              // 刷新文件列表
              that.getUploadedFiles()
            } else {
              wx.showToast({ title: deleteRes.data.msg || '删除失败', icon: 'none' })
            }
          } catch (err) {
            wx.showToast({ title: '删除失败，请重试', icon: 'none' })
            console.error('文件删除失败：', err)
          }
        }
      }
    })
  }
```

## 16. pages/draw/draw.js（AI画图页面逻辑）

```javascript
// pages/draw/draw.js
const app = getApp()
Page({
  data: {
    prompt: '', // 画图描述词
    size: '1024x1024', // 图片尺寸
    imageUrl: '', // 生成的图片URL
    isLoading: false, // 加载状态
    sizeOptions: [
      { name: '1024x1024', value: '1024x1024' },
      { name: '1024x1536', value: '1024x1536' },
      { name: '1536x1024', value: '1536x1024' }
    ]
  },

  // 输入框内容变化（描述词）
  onPromptInput(e) {
    this.setData({ prompt: e.detail.value })
  },

  // 选择图片尺寸
  bindSizeChange(e) {
    this.setData({ size: e.detail.value })
  },

  // 生成图片
  async generateImage() {
    const { prompt, size, isLoading } = this.data
    if (!prompt.trim()) {
      wx.showToast({ title: '请输入画图描述词', icon: 'none' })
      return
    }
    if (isLoading) {
      wx.showToast({ title: '请等待图片生成完成', icon: 'none' })
      return
    }

    this.setData({ isLoading: true, imageUrl: '' })
    try {
      const res = await wx.request({
        url: `${app.globalData.baseUrl}/draw`,
        method: 'POST',
        header: {
          'Authorization': `Bearer ${app.globalData.token}`,
          'Content-Type': 'application/json'
        },
        data: {
          prompt: prompt.trim(),
          size: size
        }
      })

      if (res.data.code === 200) {
        this.setData({ imageUrl: res.data.data.image_url })
        wx.showToast({ title: '图片生成成功' })
      } else {
        wx.showToast({ title: res.data.msg || '图片生成失败', icon: 'none' })
      }
    } catch (err) {
      wx.showToast({ title: '网络异常，请重试', icon: 'none' })
      console.error('图片生成失败：', err)
    } finally {
      this.setData({ isLoading: false })
    }
  },

  // 预览图片
  previewImage() {
    const { imageUrl } = this.data
    if (!imageUrl) {
      wx.showToast({ title: '暂无图片可预览', icon: 'none' })
      return
    }
    wx.previewImage({
      urls: [imageUrl],
      current: imageUrl
    })
  },

  // 保存图片到相册
  saveImage() {
    const { imageUrl } = this.data
    if (!imageUrl) {
      wx.showToast({ title: '暂无图片可保存', icon: 'none' })
      return
    }

    // 先授权相册
    wx.getSetting({
      success(res) {
        if (!res.authSetting['scope.writePhotosAlbum']) {
          wx.authorize({
            scope: 'scope.writePhotosAlbum',
            success() {
              // 授权成功，保存图片
              wx.downloadFile({
                url: imageUrl,
                success(downloadRes) {
                  if (downloadRes.statusCode === 200) {
                    wx.saveImageToPhotosAlbum({
                      filePath: downloadRes.tempFilePath,
                      success() {
                        wx.showToast({ title: '图片保存成功' })
                      },
                      fail(err) {
                        wx.showToast({ title: '图片保存失败', icon: 'none' })
                        console.error('保存图片失败：', err)
                      }
                    })
                  }
                },
                fail(err) {
                  wx.showToast({ title: '图片下载失败', icon: 'none' })
                  console.error('下载图片失败：', err)
                }
              })
            },
            fail() {
              wx.showToast({ title: '请授权相册权限', icon: 'none' })
              wx.openSetting({
                success(res) {
                  console.log('授权设置：', res.authSetting)
                }
              })
            }
          })
        } else {
          // 已授权，直接保存图片
          wx.downloadFile({
            url: imageUrl,
            success(downloadRes) {
              if (downloadRes.statusCode === 200) {
                wx.saveImageToPhotosAlbum({
                  filePath: downloadRes.tempFilePath,
                  success() {
                    wx.showToast({ title: '图片保存成功' })
                  },
                  fail(err) {
                    wx.showToast({ title: '图片保存失败', icon: 'none' })
                    console.error('保存图片失败：', err)
                  }
                })
              }
            },
            fail(err) {
              wx.showToast({ title: '图片下载失败', icon: 'none' })
              console.error('下载图片失败：', err)
            }
          })
        }
      }
    })
  }
```
