# Mixture of Experts (MoE)：大规模模型的稀疏化架构

## 一、MoE 的核心思想与数学基础

### 1.1 基本概念

**Mixture of Experts (MoE)** 是一种**条件计算（Conditional Computation）** 技术，核心思想是：对于每个输入样本，只激活网络中的一小部分参数进行计算。

传统稠密模型：
$$
y = f_{\text{all}}(x) \quad \text{（使用所有参数）}
$$

MoE 模型：
$$
y = \sum_{i=1}^N g_i(x) \cdot f_i(x) \quad \text{（只使用部分专家）}
$$
其中 $g_i(x)$ 是门控函数，决定专家 $f_i(x)$ 的权重。

### 1.2 数学形式化定义

设有 $N$ 个专家 $\{E_1, E_2, \dots, E_N\}$，每个专家是一个函数 $E_i: \mathbb{R}^d \to \mathbb{R}^d$。

**门控网络** $G: \mathbb{R}^d \to \mathbb{R}^N$：
$$
G(x) = \text{softmax}(W_g x + b_g) \quad \text{或更复杂的路由机制}
$$

**稀疏门控**：只选择 top-k 个专家：
$$
\tilde{G}(x) = \text{TopK}(\text{softmax}(W_g x), k)
$$

**MoE 层输出**：
$$
\text{MoE}(x) = \sum_{i=1}^N \tilde{g}_i(x) \cdot E_i(x)
$$
其中 $\tilde{g}_i(x)$ 是稀疏化后的门控权重。

## 二、MoE 在 Transformer 中的实现

### 2.1 标准 MoE Transformer 层

```python
class MoETransformerLayer(nn.Module):
    def __init__(self, d_model, d_ff, num_experts, top_k=2):
        super().__init__()
        # 自注意力层（共享）
        self.attention = MultiHeadAttention(d_model)
        
        # MoE 前馈层（替换传统的FFN）
        self.moe = MoELayer(
            d_model=d_model,
            d_ff=d_ff,
            num_experts=num_experts,
            top_k=top_k
        )
        
        # 层归一化
        self.norm1 = nn.LayerNorm(d_model)
        self.norm2 = nn.LayerNorm(d_model)
    
    def forward(self, x):
        # 自注意力
        attn_output = self.attention(x)
        x = self.norm1(x + attn_output)
        
        # MoE 前馈
        moe_output = self.moe(x)
        x = self.norm2(x + moe_output)
        
        return x
```

### 2.2 MoE 层详细实现

```python
class MoELayer(nn.Module):
    def __init__(self, d_model, d_ff, num_experts, top_k=2, 
                 capacity_factor=1.0, noisy_gating=True):
        super().__init__()
        self.d_model = d_model
        self.d_ff = d_ff
        self.num_experts = num_experts
        self.top_k = top_k
        self.capacity_factor = capacity_factor
        
        # 专家网络：每个专家是一个独立的FFN
        self.experts = nn.ModuleList([
            FeedForward(d_model, d_ff) for _ in range(num_experts)
        ])
        
        # 门控网络
        self.gate = nn.Linear(d_model, num_experts)
        
        # 噪声层（用于负载均衡）
        if noisy_gating:
            self.w_noise = nn.Linear(d_model, num_experts)
        
        # 辅助损失权重
        self.aux_loss_weight = 0.01
        
    def forward(self, x):
        """x: [batch_size, seq_len, d_model]"""
        batch_size, seq_len, d_model = x.shape
        x_flat = x.reshape(-1, d_model)  # [batch*seq, d_model]
        
        # 1. 计算门控权重
        gates = self.gate(x_flat)  # [batch*seq, num_experts]
        
        if self.noisy_gating and self.training:
            # 添加可学习噪声，促进负载均衡
            noise = torch.randn_like(gates) * F.softplus(self.w_noise(x_flat))
            gates = gates + noise
        
        # 2. Top-k 稀疏化
        top_k_vals, top_k_indices = torch.topk(
            gates, k=self.top_k, dim=-1, sorted=False
        )
        
        # 3. softmax 归一化（只在top-k上）
        top_k_vals = F.softmax(top_k_vals, dim=-1)
        
        # 4. 创建掩码和one-hot编码
        masks = F.one_hot(top_k_indices, num_classes=self.num_experts).float()
        # masks: [batch*seq, num_experts]，每个位置有k个1
        
        # 5. 计算每个token分配的专家
        expert_counts = masks.sum(dim=0)  # [num_experts]
        
        # 6. 计算容量（每个专家最多处理的token数）
        capacity = int(self.capacity_factor * batch_size * seq_len / self.num_experts)
        capacity = max(capacity, 1)
        
        # 7. 负载均衡路由（限制每个专家的token数）
        sorted_indices = torch.argsort(gates, dim=-1, descending=True)
        expert_assignments = []
        
        # 为每个token分配专家，考虑容量限制
        expert_token_count = torch.zeros(self.num_experts, device=x.device)
        
        for i in range(batch_size * seq_len):
            token_indices = sorted_indices[i]
            for expert_idx in token_indices:
                if expert_token_count[expert_idx] < capacity:
                    expert_assignments.append(expert_idx)
                    expert_token_count[expert_idx] += 1
                    break
        
        expert_assignments = torch.tensor(expert_assignments, device=x.device)
        
        # 8. 分派tokens到专家
        outputs = torch.zeros_like(x_flat)
        aux_loss = 0.0
        
        for expert_idx in range(self.num_experts):
            # 获取分配给当前专家的tokens
            expert_mask = (expert_assignments == expert_idx)
            if expert_mask.sum() == 0:
                continue
            
            # 获取这些tokens的输入
            expert_input = x_flat[expert_mask]
            
            # 专家前向传播
            expert_output = self.expertsexpert_input
            
            # 获取这些tokens的门控权重
            token_gates = top_k_vals[expert_mask]
            # 找到当前专家在top-k中的位置
            for k in range(self.top_k):
                expert_at_k = (top_k_indices[expert_mask, k] == expert_idx)
                if expert_at_k.any():
                    # 应用门控权重
                    outputs[expert_mask] += (
                        expert_output * token_gates[expert_at_k, k:k+1]
                    )
        
        # 9. 计算辅助损失（负载均衡损失）
        if self.training:
            aux_loss = self._auxiliary_loss(expert_counts, batch_size * seq_len)
        
        outputs = outputs.reshape(batch_size, seq_len, d_model)
        
        return outputs, aux_loss
    
    def _auxiliary_loss(self, expert_counts, total_tokens):
        """
        负载均衡损失：鼓励均匀分配tokens给专家
        """
        # 每个专家处理的token比例
        expert_fraction = expert_counts / total_tokens
        
        # 理想均匀分布
        uniform_fraction = torch.ones_like(expert_fraction) / self.num_experts
        
        # 使用KL散度或均方误差
        aux_loss = F.mse_loss(expert_fraction, uniform_fraction)
        
        return aux_loss * self.aux_loss_weight
```

## 三、MoE 的关键技术细节

### 3.1 门控机制变体

#### **Top-k 门控**（最常用）
```python
def top_k_gating(x, W_gate, k=2, capacity_factor=1.0):
    """
    x: [batch_size, seq_len, d_model]
    返回: 稀疏门控权重和专家分配
    """
    logits = x @ W_gate.T  # [batch*seq, num_experts]
    
    # Top-k选择
    top_k_logits, top_k_indices = torch.topk(logits, k=k)
    
    # softmax归一化
    top_k_gates = F.softmax(top_k_logits, dim=-1)
    
    return top_k_gates, top_k_indices
```

#### **Noisy Top-k 门控**（Google Switch Transformer）
```python
def noisy_top_k_gating(x, W_gate, W_noise, k=2, noise_epsilon=1e-2):
    """
    添加噪声促进探索
    """
    clean_logits = x @ W_gate.T
    
    # 添加可学习噪声
    noise = torch.randn_like(clean_logits) * F.softplus(x @ W_noise.T)
    noisy_logits = clean_logits + noise_epsilon * noise
    
    # 保持原始logits用于前向传播
    top_k_logits, top_k_indices = torch.topk(noisy_logits, k=k)
    top_k_gates = F.softmax(clean_logits.gather(1, top_k_indices), dim=-1)
    
    return top_k_gates, top_k_indices
```

#### **Hash 门控**（简单但有效）
```python
def hash_gating(x, num_experts, k=2):
    """
    基于token哈希分配专家，完全确定性的
    """
    batch_size, seq_len, d_model = x.shape
    x_flat = x.reshape(-1, d_model)
    
    # 使用token ID的哈希值
    token_ids = torch.arange(batch_size * seq_len, device=x.device)
    
    # 为每个token分配k个专家
    assignments = []
    for i in range(k):
        # 使用不同的哈希函数
        hash_values = torch.remainder(
            token_ids * (i + 1), 
            num_experts
        )
        assignments.append(hash_values)
    
    assignments = torch.stack(assignments, dim=1)  # [batch*seq, k]
    
    # 均匀权重
    gates = torch.ones(batch_size * seq_len, k, device=x.device) / k
    
    return gates, assignments
```

### 3.2 负载均衡策略

#### **容量因子（Capacity Factor）**
```python
class CapacityScheduler:
    """
    动态调整专家容量，平衡负载和计算效率
    """
    def __init__(self, init_capacity=1.0, min_capacity=0.5, 
                 max_capacity=2.0, warmup_steps=1000):
        self.capacity = init_capacity
        self.min_capacity = min_capacity
        self.max_capacity = max_capacity
        self.warmup_steps = warmup_steps
        self.step_count = 0
        
    def update(self, expert_utilization):
        """
        expert_utilization: 每个专家的使用率 [num_experts]
        """
        self.step_count += 1
        
        if self.step_count < self.warmup_steps:
            # 预热阶段：线性增加容量
            self.capacity = min(
                self.max_capacity,
                self.min_capacity + 
                (self.max_capacity - self.min_capacity) * 
                self.step_count / self.warmup_steps
            )
        else:
            # 基于使用率调整
            avg_utilization = expert_utilization.mean().item()
            
            if avg_utilization > 0.9:
                # 使用率过高，增加容量
                self.capacity = min(self.max_capacity, self.capacity * 1.01)
            elif avg_utilization < 0.5:
                # 使用率过低，减少容量
                self.capacity = max(self.min_capacity, self.capacity * 0.99)
        
        return self.capacity
```

#### **辅助损失函数**

```python
def load_balancing_loss(expert_counts, total_tokens, num_experts):
    """
    专家负载均衡损失
    """
    # 每个专家的token比例
    fraction = expert_counts / total_tokens
    
    # 理想的均匀分布
    uniform = torch.ones(num_experts, device=expert_counts.device) / num_experts
    
    # 使用KL散度
    kl_div = F.kl_div(
        fraction.log(), 
        uniform, 
        reduction='batchmean'
    )
    
    # 或者使用CV（变异系数）作为损失
    mean = fraction.mean()
    std = fraction.std()
    cv_loss = std / (mean + 1e-6)  # 变异系数
    
    return kl_div + 0.1 * cv_loss
```

### 3.3 专家专业化

```python
class ExpertSpecialization(nn.Module):
    """
    鼓励专家专业化的机制
    """
    def __init__(self, num_experts, d_model, specialization_dim=64):
        super().__init__()
        
        # 每个专家的专业化向量
        self.specialization = nn.Parameter(
            torch.randn(num_experts, specialization_dim)
        )
        
        # 输入到专业化空间的投影
        self.input_proj = nn.Linear(d_model, specialization_dim)
        
    def compute_specialization_scores(self, x):
        """
        计算输入与每个专家的匹配分数
        """
        # 将输入投影到专业化空间
        x_proj = self.input_proj(x)  # [batch*seq, spec_dim]
        
        # 计算与每个专家的余弦相似度
        spec_norm = F.normalize(self.specialization, dim=-1)  # [num_experts, spec_dim]
        x_proj_norm = F.normalize(x_proj, dim=-1)  # [batch*seq, spec_dim]
        
        # 相似度矩阵 [batch*seq, num_experts]
        similarity = x_proj_norm @ spec_norm.T
        
        return similarity
    
    def specialization_loss(self, expert_assignments, x):
        """
        专业化损失：鼓励专家处理与其专业化匹配的token
        """
        batch_size_seq = x.shape[0]
        
        # 计算相似度
        similarity = self.compute_specialization_scores(x)  # [batch*seq, num_experts]
        
        # 获取每个token分配给的专家
        expert_mask = F.one_hot(
            expert_assignments, 
            num_classes=self.num_experts
        ).float()  # [batch*seq, num_experts]
        
        # 最大化分配专家与token的相似度
        assigned_similarity = (similarity * expert_mask).sum(dim=-1)
        
        # 损失：最小化负相似度
        loss = -assigned_similarity.mean()
        
        return loss
```

## 四、MoE 的分布式训练

### 4.1 专家并行（Expert Parallelism）

```python
class ExpertParallelMoE(nn.Module):
    """
    将专家分布在不同设备上
    """
    def __init__(self, d_model, d_ff, num_experts, top_k=2, 
                 num_devices=None):
        super().__init__()
        
        self.num_experts = num_experts
        self.top_k = top_k
        self.d_model = d_model
        
        # 确定设备数量
        if num_devices is None:
            num_devices = torch.cuda.device_count()
        self.num_devices = num_devices
        
        # 计算每个设备的专家数
        experts_per_device = num_experts // num_devices
        assert num_experts % num_devices == 0, "专家数必须能被设备数整除"
        
        # 在每个设备上创建专家
        self.experts = nn.ModuleList()
        self.expert_device_map = {}
        
        for device_idx in range(num_devices):
            start_idx = device_idx * experts_per_device
            end_idx = start_idx + experts_per_device
            
            # 在对应设备上创建专家
            device_experts = nn.ModuleList([
                FeedForward(d_model, d_ff).to(f"cuda:{device_idx}")
                for _ in range(experts_per_device)
            ])
            
            self.experts.append(device_experts)
            
            # 记录专家到设备的映射
            for expert_idx in range(start_idx, end_idx):
                self.expert_device_map[expert_idx] = device_idx
        
        # 门控网络（在第一个设备上）
        self.gate = nn.Linear(d_model, num_experts).to("cuda:0")
        
        # 用于跨设备通信
        self.world_size = num_devices
        self.rank = torch.distributed.get_rank() if torch.distributed.is_initialized() else 0
    
    def forward(self, x):
        """
        分布式专家前向传播
        """
        batch_size, seq_len, d_model = x.shape
        x_flat = x.reshape(-1, d_model)
        
        # 1. 门控计算（在设备0上）
        if self.rank == 0:
            gates = self.gate(x_flat)
            top_k_gates, top_k_indices = torch.topk(gates, k=self.top_k)
            top_k_gates = F.softmax(top_k_gates, dim=-1)
            
            # 广播门控结果
            gate_info = (top_k_gates, top_k_indices)
        else:
            gate_info = (None, None)
        
        # 广播门控信息到所有设备
        if torch.distributed.is_initialized():
            torch.distributed.broadcast(gate_info, src=0)
        
        top_k_gates, top_k_indices = gate_info
        
        # 2. 根据门控结果路由tokens
        # 首先收集所有需要处理的专家
        all_experts_needed = top_k_indices.unique()
        
        # 按设备分组
        device_to_tokens = {i: [] for i in range(self.num_devices)}
        device_to_gates = {i: [] for i in range(self.num_devices)}
        device_to_indices = {i: [] for i in range(self.num_devices)}
        
        for token_idx in range(batch_size * seq_len):
            for k in range(self.top_k):
                expert_idx = top_k_indices[token_idx, k].item()
                device_idx = self.expert_device_map[expert_idx]
                
                device_to_tokens[device_idx].append(x_flat[token_idx])
                device_to_gates[device_idx].append(top_k_gates[token_idx, k])
                device_to_indices[device_idx].append((token_idx, expert_idx))
        
        # 3. 在每个设备上并行处理
        outputs = torch.zeros_like(x_flat)
        
        for device_idx in range(self.num_devices):
            if not device_to_tokens[device_idx]:
                continue
            
            # 将数据移动到对应设备
            device_tokens = torch.stack(device_to_tokens[device_idx]).to(f"cuda:{device_idx}")
            device_gates = torch.tensor(device_to_gates[device_idx], 
                                       device=f"cuda:{device_idx}")
            device_indices = device_to_indices[device_idx]
            
            # 处理本设备上的所有专家
            expert_outputs = {}
            
            for token_batch, gate_val, (token_idx, expert_idx) in zip(
                device_tokens, device_gates, device_indices
            ):
                if expert_idx not in expert_outputs:
                    # 计算专家输出（批处理）
                    local_expert_idx = expert_idx % (self.num_experts // self.num_devices)
                    expert = self.experts[device_idx][local_expert_idx]
                    
                    # 收集所有需要该专家的tokens
                    mask = torch.tensor([
                        idx == expert_idx for _, idx in device_indices
                    ], device=f"cuda:{device_idx}")
                    
                    if mask.any():
                        batch_tokens = device_tokens[mask]
                        batch_output = expert(batch_tokens)
                        expert_outputs[expert_idx] = (batch_output, mask)
                
                # 应用门控权重
                batch_output, mask = expert_outputs[expert_idx]
                token_pos_in_batch = mask.nonzero()[0].item()
                output_val = batch_output[token_pos_in_batch] * gate_val
                
                # 收集到输出（需要跨设备通信）
                if device_idx == self.rank:
                    outputs[token_idx] += output_val.to(outputs.device)
                else:
                    # 发送到拥有该token的设备
                    # 简化：这里假设所有输出最终汇集到设备0
                    pass
        
        # 4. 跨设备汇总结果
        if torch.distributed.is_initialized():
            # All-reduce汇总所有设备的输出
            torch.distributed.all_reduce(outputs, op=torch.distributed.ReduceOp.SUM)
        
        outputs = outputs.reshape(batch_size, seq_len, d_model)
        
        return outputs
```

### 4.2 数据并行中的MoE

```python
class DataParallelMoE(nn.Module):
    """
    在数据并行中处理MoE
    """
    def __init__(self, moe_layer, device_ids=None):
        super().__init__()
        self.moe_layer = moe_layer
        self.device_ids = device_ids or list(range(torch.cuda.device_count()))
        
    def forward(self, *inputs):
        """
        数据并行前向传播
        """
        # 复制输入到各个设备
        scattered_inputs = scatter(inputs, self.device_ids)
        
        # 在各个设备上并行计算
        replicas = self._replicate(self.moe_layer, self.device_ids)
        outputs = parallel_apply(replicas, scattered_inputs)
        
        # 收集输出
        output = gather(outputs, self.device_ids[0])
        
        return output
    
    def _replicate(self, module, device_ids):
        """
        复制MoE层到各个设备
        注意：需要处理门控网络的一致性
        """
        replicas = []
        for device_id in device_ids:
            replica = copy.deepcopy(module)
            replica.to(f"cuda:{device_id}")
            replicas.append(replica)
        
        # 同步门控网络的参数
        with torch.no_grad():
            for param_name, param in module.gate.named_parameters():
                param_data = param.data
                for replica in replicas:
                    getattr(replica.gate, param_name).data.copy_(param_data)
        
        return replicas
```

### 5.1 动态批处理与MoE

```python
class DynamicBatchingMoE:
    """
    MoE的动态批处理优化
    """
    def __init__(self, moe_layer, max_batch_size=32, 
                 expert_capacity=512):
        self.moe_layer = moe_layer
        self.max_batch_size = max_batch_size
        self.expert_capacity = expert_capacity
        
        # 专家队列：为每个专家维护待处理tokens
        self.expert_queues = {
            i: deque(maxlen=expert_capacity) 
            for i in range(moe_layer.num_experts)
        }
        
        # 批处理调度器
        self.scheduler = ExpertScheduler(moe_layer.num_experts)
    
    def process_batch(self, x):
        """
        动态批处理MoE计算
        """
        batch_size, seq_len, d_model = x.shape
        x_flat = x.reshape(-1, d_model)
        
        # 1. 计算门控
        gates = self.moe_layer.gate(x_flat)
        top_k_gates, top_k_indices = torch.topk(
            gates, k=self.moe_layer.top_k
        )
        top_k_gates = F.softmax(top_k_gates, dim=-1)
        
        # 2. 将tokens分配到专家队列
        token_assignments = {}
        for token_idx in range(batch_size * seq_len):
            for k in range(self.moe_layer.top_k):
                expert_idx = top_k_indices[token_idx, k].item()
                gate_val = top_k_gates[token_idx, k].item()
                
                token_info = {
                    'data': x_flat[token_idx],
                    'gate': gate_val,
                    'token_idx': token_idx,
                    'k_idx': k
                }
                
                self.expert_queues[expert_idx].append(token_info)
                if expert_idx not in token_assignments:
                    token_assignments[expert_idx] = []
                token_assignments[expert_idx].append(token_info)
        
        # 3. 调度专家计算
        outputs = torch.zeros_like(x_flat)
        
        while True:
            # 选择下一个要计算的专家
            expert_idx = self.scheduler.next_expert()
            if expert_idx is None:
                break
            
            # 从队列中取出tokens（最多max_batch_size个）
            batch_tokens = []
            batch_gates = []
            batch_info = []
            
            while (len(batch_tokens) < self.max_batch_size and 
                   self.expert_queues[expert_idx]):
                token_info = self.expert_queues[expert_idx].popleft()
                batch_tokens.append(token_info['data'])
                batch_gates.append(token_info['gate'])
                batch_info.append(token_info)
            
            if not batch_tokens:
                continue
            
            # 批处理计算
            batch_tokens = torch.stack(batch_tokens)
            expert_output = self.moe_layer.expertsbatch_tokens
            
            # 应用门控权重
            for i, token_info in enumerate(batch_info):
                output_val = expert_output[i] * batch_gates[i]
                outputs[token_info['token_idx']] += output_val
        
        return outputs.reshape(batch_size, seq_len, d_model)
```

```python
class MoEKVCache:
    """
    MoE模型专用的KV Cache优化
    """
    def __init__(self, num_experts, d_model, num_heads, head_dim):
        self.num_experts = num_experts
        self.d_model = d_model
        
        # 为每个专家维护独立的KV Cache
        self.kv_caches = [
            {
                'keys': None,
                'values': None,
                'last_used': 0  # LRU时间戳
            }
            for _ in range(num_experts)
        ]
        
        # 专家使用统计
        self.expert_usage = torch.zeros(num_experts)
        
    def update_cache(self, expert_idx, new_keys, new_values):
        """
        更新指定专家的KV Cache
        """
        cache = self.kv_caches[expert_idx]
        
        if cache['keys'] is None:
            cache['keys'] = new_keys
            cache['values'] = new_values
        else:
            # 拼接新的K,V
            cache['keys'] = torch.cat([cache['keys'], new_keys], dim=2)
            cache['values'] = torch.cat([cache['values'], new_values], dim=2)
        
        # 更新使用时间
        cache['last_used'] = time.time()
        self.expert_usage[expert_idx] += new_keys.shape[2]  # 序列长度
    
    def get_cache(self, expert_idx, seq_indices=None):
        """
        获取专家的KV Cache
        """
        cache = self.kv_caches[expert_idx]
        
        if seq_indices is None:
            return cache['keys'], cache['values']
        else:
            if cache['keys'] is None:
                return None, None
            else:
                return (cache['keys'][:, :, seq_indices], 
                        cache['values'][:, :, seq_indices])
    
    def prune_cache(self, max_tokens_per_expert):
        """
        修剪缓存，移除最少使用的部分
        """
        for expert_idx in range(self.num_experts):
            cache = self.kv_caches[expert_idx]
            if cache['keys'] is not None:
                seq_len = cache['keys'].shape[2]
                if seq_len > max_tokens_per_expert:
                    # 保留最近使用的tokens
                    keep_indices = torch.arange(
                        seq_len - max_tokens_per_expert, seq_len
                    )
                    cache['keys'] = cache['keys'][:, :, keep_indices]
                    cache['values'] = cache['values'][:, :, keep_indices]
```

```python
class SwitchTransformerLayer(nn.Module):
    """
    Switch Transformer: 只选择1个专家 (top-1 gating)
    比top-2更高效，计算量减半
    """
    def __init__(self, d_model, d_ff, num_experts, 
                 capacity_factor=1.25, drop_tokens=False):
        super().__init__()
        self.num_experts = num_experts
        self.capacity_factor = capacity_factor
        self.drop_tokens = drop_tokens
        
        # 专家
        self.experts = nn.ModuleList([
            FeedForward(d_model, d_ff * 4)  # 更大的FFN
            for _ in range(num_experts)
        ])
        
        # 门控（top-1）
        self.gate = nn.Linear(d_model, num_experts)
        
        # 用于负载均衡的辅助损失
        self.aux_loss = 0.0
    
    def forward(self, x):
        batch_size, seq_len, d_model = x.shape
        x_flat = x.reshape(-1, d_model)
        
        # 门控计算
        gates = self.gate(x_flat)  # [batch*seq, num_experts]
        
        # 选择top-1专家
        expert_weights, expert_indices = torch.max(gates, dim=-1)
        expert_weights = torch.softmax(expert_weights, dim=0)
        
        # 创建专家掩码
        expert_mask = F.one_hot(expert_indices, 
                               num_classes=self.num_experts).float()
        
        # 计算容量
        capacity = int(self.capacity_factor * 
                      batch_size * seq_len / self.num_experts)
        
        # 路由逻辑
        outputs = torch.zeros_like(x_flat)
        
        for expert_idx in range(self.num_experts):
            # 获取分配给该专家的tokens
            idx = (expert_indices == expert_idx).nonzero(as_tuple=True)[0]
            
            if len(idx) == 0:
                continue
            
            # 如果超过容量，选择权重最高的tokens
            if len(idx) > capacity:
                if self.drop_tokens:
                    # 丢弃低权重的tokens
                    top_idx = idx[expert_weights[idx].argsort(descending=True)[:capacity]]
                else:
                    # 或者均匀采样
                    top_idx = idx[torch.randperm(len(idx))[:capacity]]
                idx = top_idx
            
            if len(idx) == 0:
                continue
            
            # 专家计算
            expert_input = x_flat[idx]
            expert_output = self.expertsexpert_input
            
            # 加权输出
            weights = expert_weights[idx].unsqueeze(-1)
            outputs[idx] = expert_output * weights
        
        # 计算辅助损失
        if self.training:
            self.aux_loss = self._load_balancing_loss(
                expert_mask.sum(dim=0), 
                batch_size * seq_len
            )
        
        return outputs.reshape(batch_size, seq_len, d_model)
```

### 6.2 GShard（Google）

```python
class GShardMoE(nn.Module):
    """
    GShard: 针对大规模训练的MoE优化
    特点：
    1. 专家容量因子动态调整
    2. 专家间梯度共享
    3. 通信优化
    """
    def __init__(self, d_model, d_ff, num_experts, 
                 top_k=2, capacity_factor=2.0,
                 min_capacity=4):
        super().__init__()
        self.num_experts = num_experts
        self.top_k = top_k
        self.min_capacity = min_capacity
        
        # 专家网络
        self.experts = nn.ModuleList([
            nn.Sequential(
                nn.Linear(d_model, d_ff),
                nn.GELU(),
                nn.Linear(d_ff, d_model)
            )
            for _ in range(num_experts)
        ])
        
        # 门控网络
        self.gate = nn.Linear(d_model, num_experts, bias=False)
        
        # 梯度缩放（用于平衡专家）
        self.gradient_scale = 1.0 / (num_experts * top_k)
        
        # 容量调度器
        self.capacity_scheduler = CapacityScheduler(
            init_capacity=capacity_factor,
            min_capacity=1.0,
            max_capacity=4.0
        )
    
    def forward(self, x):
        batch_size, seq_len, d_model = x.shape
        x_flat = x.reshape(-1, d_model)
        
        # 计算门控logits
        logits = self.gate(x_flat)  # [batch*seq, num_experts]
        
        # 添加随机噪声（训练时）
        if self.training:
            noise = torch.randn_like(logits) / (num_experts ** 0.5)
            logits = logits + noise
        
        # Top-k选择
        gates, indices = torch.topk(logits, k=self.top_k, dim=-1)
        gates = F.softmax(gates, dim=-1)
        
        # 计算动态容量
        capacity = self.capacity_scheduler.update(
            self._expert_utilization(indices)
        )
        capacity = max(int(capacity * batch_size * seq_len / self.num_experts), 
                      self.min_capacity)
        
        # 创建调度掩码
        mask = self._create_routing_mask(indices, capacity)
        
        # 分派tokens到专家
        outputs = self._dispatch_and_compute(x_flat, gates, indices, mask)
        
        # 梯度缩放
        if self.training:
            outputs = outputs * self.gradient_scale
        
        return outputs.reshape(batch_size, seq_len, d_model)
    
    def _create_routing_mask(self, indices, capacity):
        """
        GShard的智能路由掩码
        """
        batch_seq = indices.shape[0]
        mask = torch.zeros(batch_seq, self.num_experts, 
                          device=indices.device, dtype=torch.bool)
        
        # 为每个专家计数
        expert_count = torch.zeros(self.num_experts, device=indices.device)
        
        # 第一轮：为每个token分配首选专家
        for i in range(batch_seq):
            primary_expert = indices[i, 0].item()
            if expert_count[primary_expert] < capacity:
                mask[i, primary_expert] = True
                expert_count[primary_expert] += 1
        
        # 第二轮：为未分配的token分配备用专家
        for i in range(batch_seq):
            if not mask[i].any():  # 未分配
                for k in range(1, self.top_k):
                    backup_expert = indices[i, k].item()
                    if expert_count[backup_expert] < capacity:
                        mask[i, backup_expert] = True
                        expert_count[backup_expert] += 1
                        break
        
        return mask
    
    def _dispatch_and_compute(self, x, gates, indices, mask):
        """
        分派tokens并计算专家输出
        """
        outputs = torch.zeros_like(x)
        
        for expert_idx in range(self.num_experts):
            # 获取分配给该专家的tokens
            token_indices = mask[:, expert_idx].nonzero(as_tuple=True)[0]
            
            if len(token_indices) == 0:
                continue
            
            # 获取这些tokens的输入和门控权重
            expert_input = x[token_indices]
            
            # 计算专家输出
            expert_output = self.expertsexpert_input
            
            # 应用门控权重
            for i, token_idx in enumerate(token_indices):
                # 找到该token对应当前专家的门控权重
                for k in range(self.top_k):
                    if indices[token_idx, k] == expert_idx:
                        gate_val = gates[token_idx, k]
                        break
                
                outputs[token_idx] += expert_output[i] * gate_val
        
        return outputs
```

### 6.3 DeepSeek-MoE（深度求索）

```python
class DeepSeekMoE(nn.Module):
    """
    DeepSeek-MoE: 针对推理优化的MoE架构
    特点：
    1. 细粒度专家（更多但更小的专家）
    2. 共享专家和路由专家分离
    3. 动态专家激活
    """
    def __init__(self, d_model, d_ff, num_experts=64, 
                 num_shared_experts=4, num_routed_experts=8,
                 top_k=2, capacity_factor=1.0):
        super().__init__()
        
        self.d_model = d_model
        self.num_experts = num_experts
        self.num_shared_experts = num_shared_experts
        self.num_routed_experts = num_routed_experts
        self.top_k = top_k
        
        # 共享专家（总是激活）
        self.shared_experts = nn.ModuleList([
            FeedForward(d_model, d_ff // 2)  # 较小的FFN
            for _ in range(num_shared_experts)
        ])
        
        # 路由专家（条件激活）
        self.routed_experts = nn.ModuleList([
            FeedForward(d_model, d_ff)
            for _ in range(num_experts - num_shared_experts)
        ])
        
        # 门控网络
        self.gate = nn.Sequential(
            nn.Linear(d_model, d_model * 2),
            nn.GELU(),
            nn.Linear(d_model * 2, num_experts)
        )
        
        # 共享专家权重（学习每个共享专家的贡献）
        self.shared_weights = nn.Parameter(
            torch.ones(num_shared_experts) / num_shared_experts
        )
        
        # 动态专家选择
        self.expert_importance = nn.Parameter(torch.zeros(num_experts))
        
    def forward(self, x):
        batch_size, seq_len, d_model = x.shape
        x_flat = x.reshape(-1, d_model)
        
        # 1. 计算门控权重和专家重要性
        gate_logits = self.gate(x_flat)  # [batch*seq, num_experts]
        
        # 结合专家重要性（鼓励专业化）
        gate_logits = gate_logits + self.expert_importance.unsqueeze(0)
        
        # 2. 选择要激活的专家
        # 总是激活共享专家
        shared_indices = torch.arange(self.num_shared_experts, 
                                     device=x.device)
        
        # 选择top-k个路由专家
        routed_gates = gate_logits[:, self.num_shared_experts:]
        routed_top_k = min(self.top_k, self.num_routed_experts)
        
        routed_values, routed_indices = torch.topk(
            routed_gates, k=routed_top_k, dim=-1
        )
        routed_indices = routed_indices + self.num_shared_experts
        
        # 合并专家索引
        all_indices = torch.cat([
            shared_indices.repeat(batch_size * seq_len, 1),
            routed_indices
        ], dim=-1)
        
        # 合并门控值
        shared_values = gate_logits[:, :self.num_shared_experts]
        all_values = torch.cat([shared_values, routed_values], dim=-1)
        all_values = F.softmax(all_values, dim=-1)
        
        # 3. 计算共享专家输出（总是计算）
        shared_output = torch.zeros_like(x_flat)
        for i in range(self.num_shared_experts):
            expert_output = self.shared_expertsx_flat
            weight = self.shared_weights[i] * all_values[:, i].unsqueeze(-1)
            shared_output += expert_output * weight
        
        # 4. 计算路由专家输出（条件计算）
        routed_output = torch.zeros_like(x_flat)
        
        # 为每个选中的路由专家计算输出
        for k in range(routed_top_k):
            expert_indices = routed_indices[:, k] - self.num_shared_experts
            gate_values = all_values[:, self.num_shared_experts + k]
            
            # 收集需要当前专家的tokens
            for expert_id in range(len(self.routed_experts)):
                mask = (expert_indices == expert_id)
                if mask.any():
                    tokens = x_flat[mask]
                    expert_out = self.routed_expertstokens
                    
                    # 应用门控权重
                    weights = gate_values[mask].unsqueeze(-1)
                    routed_output[mask] += expert_out * weights
        
        # 5. 合并输出
        output = shared_output + routed_output
        
        return output.reshape(batch_size, seq_len, d_model)
```

### 6.4 Mixtral 8x7B（Mistral AI）

```python
class MixtralMoE(nn.Module):
    """
    Mixtral 8x7B: 稀疏MoE模型，8个专家选2个
    特点：
    1. 每个专家是完整的FFN
    2. 路由器使用softmax温度调节
    3. 专家dropout防止过拟合
    """
    def __init__(self, d_model, d_ff, num_experts=8, top_k=2,
                 router_temp=1.0, expert_dropout=0.1):
        super().__init__()
        
        self.num_experts = num_experts
        self.top_k = top_k
        self.router_temp = router_temp
        self.expert_dropout = expert_dropout
        
        # 专家网络（每个都是完整的FFN）
        self.experts = nn.ModuleList([
            nn.Sequential(
                nn.Linear(d_model, d_ff),
                nn.SiLU(),  # Mixtral使用SiLU激活
                nn.Linear(d_ff, d_model)
            )
            for _ in range(num_experts)
        ])
        
        # 路由器网络
        self.router = nn.Linear(d_model, num_experts, bias=False)
        
        # 专家dropout
        self.dropout = nn.Dropout(expert_dropout) if expert_dropout > 0 else None
        
        # 负载均衡损失
        self.aux_loss_weight = 0.01
        
    def forward(self, x):
        batch_size, seq_len, d_model = x.shape
        x_flat = x.reshape(-1, d_model)
        
        # 路由器logits
        router_logits = self.router(x_flat)  # [batch*seq, num_experts]
        
        # 应用温度调节
        if self.router_temp != 1.0:
            router_logits = router_logits / self.router_temp
        
        # Top-k选择
        top_k_logits, top_k_indices = torch.topk(
            router_logits, k=self.top_k, dim=-1
        )
        
        # Router softmax（只在top-k上）
        router_weights = F.softmax(top_k_logits, dim=-1)
        
        # 创建专家掩码
        expert_mask = F.one_hot(
            top_k_indices, 
            num_classes=self.num_experts
        ).float()  # [batch*seq, num_experts, top_k]
        
        # 计算每个专家应该处理的tokens数
        expert_counts = expert_mask.sum(dim=0).sum(dim=1)  # [num_experts]
        
        # 计算容量（基于负载均衡）
        capacity = self._compute_capacity(expert_counts, 
                                         batch_size * seq_len)
        
        # 最终输出
        final_output = torch.zeros_like(x_flat)
        
        # 为每个专家计算输出
        for expert_idx in range(self.num_experts):
            # 获取分配给该专家的tokens
            mask = expert_mask[:, expert_idx].any(dim=1)  # [batch*seq]
            
            if not mask.any():
                continue
            
            # 限制容量
            expert_tokens = x_flat[mask]
            if len(expert_tokens) > capacity:
                # 随机选择容量内的tokens
                indices = torch.randperm(len(expert_tokens))[:capacity]
                expert_tokens = expert_tokens[indices]
                selected_mask = torch.zeros_like(mask)
                selected_mask[mask.nonzero()[indices]] = True
                mask = selected_mask
            
            if len(expert_tokens) == 0:
                continue
            
            # 专家前向传播
            expert_out = self.expertsexpert_tokens
            
            # 应用dropout
            if self.dropout is not None and self.training:
                expert_out = self.dropout(expert_out)
            
            # 应用路由器权重
            for k in range(self.top_k):
                k_mask = expert_mask[:, expert_idx, k] & mask
                if k_mask.any():
                    # 获取这些tokens的权重
                    weights = router_weights[k_mask, k].unsqueeze(-1)
                    
                    # 加权求和
                    expert_slice = expert_out[:k_mask.sum()]
                    weighted_out = expert_slice * weights
                    
                    # 累加到最终输出
                    final_output[k_mask] += weighted_out
        
        # 计算辅助损失（负载均衡）
        aux_loss = 0
        if self.training:
            aux_loss = self._load_balancing_loss(expert_counts, 
                                                batch_size * seq_len)
        
        final_output = final_output.reshape(batch_size, seq_len, d_model)
        
        return final_output, aux_loss
    
    def _compute_capacity(self, expert_counts, total_tokens):
        """
        基于负载均衡动态计算容量
        """
        # 基本容量：平均每个专家应该处理的tokens数
        base_capacity = total_tokens / self.num_experts
        
        # 如果某个专家负载过高，增加其容量
        max_count = expert_counts.max().item()
        if max_count > base_capacity * 1.5:
            capacity = int(max_count * 1.1)  # 增加10%的缓冲
        else:
            capacity = int(base_capacity * 1.2)  # 20%的缓冲
        
        return max(capacity, 1)
    
    def _load_balancing_loss(self, expert_counts, total_tokens):
        """
        Mixtral的负载均衡损失
        """
        # 每个专家的利用率
        utilization = expert_counts / total_tokens
        
        # 理想均匀分布
        uniform = torch.ones_like(utilization) / self.num_experts
        
        # KL散度损失
        kl_loss = F.kl_div(
            utilization.log(), 
            uniform, 
            reduction='batchmean'
        )
        
        # 添加利用率方差的惩罚
        variance_penalty = utilization.var() / (uniform.var() + 1e-6)
        
        return self.aux_loss_weight * (kl_loss + 0.1 * variance_penalty)
```

### 7.1 负载均衡优化

```python
class LoadBalancingOptimizer:
    """
    MoE负载均衡优化策略
    """
    def __init__(self, moe_layers, balancing_weight=0.01):
        self.moe_layers = moe_layers
        self.balancing_weight = balancing_weight
        
        # 专家使用统计
        self.expert_usage_history = []
        
    def calculate_balancing_loss(self):
        """
        计算负载均衡损失
        """
        total_loss = 0
        for layer in self.moe_layers:
            if hasattr(layer, 'aux_loss'):
                total_loss += layer.aux_loss
        
        return total_loss * self.balancing_weight
    
    def expert_usage_stats(self):
        """
        收集专家使用统计
        """
        stats = {}
        for i, layer in enumerate(self.moe_layers):
            if hasattr(layer, 'expert_usage'):
                stats[f'layer_{i}'] = {
                    'usage': layer.expert_usage.cpu().numpy(),
                    'imbalance': self._calculate_imbalance(layer.expert_usage)
                }
        return stats
    
    def _calculate_imbalance(self, expert_usage):
        """
        计算专家使用不均衡度
        """
        usage = expert_usage.float()
        mean_usage = usage.mean()
        std_usage = usage.std()
        
        # 变异系数
        cv = std_usage / (mean_usage + 1e-6)
        
        # Gini系数
        sorted_usage, _ = torch.sort(usage)
        n = len(sorted_usage)
        index = torch.arange(1, n + 1).float()
        gini = (torch.sum((2 * index - n - 1) * sorted_usage) / 
                (n * torch.sum(sorted_usage)))
        
        return {'cv': cv.item(), 'gini': gini.item()}
    
    def adaptive_weight_adjustment(self, stats):
        """
        基于统计调整门控权重
        """
        for layer_idx, layer_stats in stats.items():
            layer = self.moe_layers[int(layer_idx.split('_')[1])]
            
            # 对于使用率低的专家，增加其门控权重偏置
            usage = torch.tensor(layer_stats['usage'])
            mean_usage = usage.mean()
            
            # 计算调整因子
            adjustment = (mean_usage / (usage + 1e-6)).clamp(0.5, 2.0)
            
            # 应用到门控网络（如果可学习）
            if hasattr(layer, 'gate'):
                with torch.no_grad():
                    # 在门控网络的偏置上进行调整
                    if hasattr(layer.gate, 'bias') and layer.gate.bias is not None:
                        layer.gate.bias.data *= adjustment
```

### 7.2 梯度裁剪与缩放

```python
class MoEGradientClipping:
    """
    MoE特有的梯度处理
    """
    def __init__(self, clip_grad_norm=1.0, 
                 expert_grad_scale=1.0,
                 router_grad_scale=10.0):
        self.clip_grad_norm = clip_grad_norm
        self.expert_grad_scale = expert_grad_scale
        self.router_grad_scale = router_grad_scale
        
    def pre_backward_hook(self, module, grad_input, grad_output):
        """
        在反向传播前调整梯度
        """
        # 对专家网络的梯度进行缩放
        if 'expert' in module._get_name().lower():
            return tuple(g * self.expert_grad_scale for g in grad_input)
        
        # 对路由器网络的梯度进行缩放
        elif 'gate' in module._get_name().lower() or 'router' in module._get_name().lower():
            return tuple(g * self.router_grad_scale for g in grad_input)
        
        return grad_input
    
    def post_backward_hook(self, parameters):
        """
        在反向传播后裁剪梯度
        """
        # 计算总梯度范数
        total_norm = 0
        for p in parameters:
            if p.grad is not None:
                param_norm = p.grad.data.norm(2)
                total_norm += param_norm.item() ** 2
        total_norm = total_norm ** 0.5
        
        # 裁剪梯度
        clip_coef = self.clip_grad_norm / (total_norm + 1e-6)
        if clip_coef < 1:
            for p in parameters:
                if p.grad is not None:
                    p.grad.data.mul_(clip_coef)
        
        return total_norm
```

```python
class MoEMetrics:
    """
    MoE模型的评估指标
    """
    @staticmethod
    def expert_utilization(expert_assignments, num_experts, seq_len):
        """
        专家利用率：每个专家处理的token比例
        """
        batch_size = expert_assignments.shape[0]
        total_tokens = batch_size * seq_len
        
        # 计算每个专家处理的token数
        expert_counts = torch.zeros(num_experts, device=expert_assignments.device)
        for i in range(num_experts):
            expert_counts[i] = (expert_assignments == i).sum()
        
        utilization = expert_counts / total_tokens
        return utilization
    
    @staticmethod
    def load_balance_score(expert_assignments, num_experts):
        """
        负载均衡分数：1表示完全均衡，0表示完全不均衡
        """
        utilization = MoEMetrics.expert_utilization(
            expert_assignments, num_experts, 
            expert_assignments.shape[1]
        )
        
        # 使用Gini系数
        sorted_util, _ = torch.sort(utilization)
        n = len(sorted_util)
        index = torch.arange(1, n + 1).float().to(sorted_util.device)
        
        gini = (torch.sum((2 * index - n - 1) * sorted_util) / 
                (n * torch.sum(sorted_util)))
        
        balance_score = 1 - gini
        return balance_score.item()
    
    @staticmethod
    def expert_specialization(expert_assignments, token_features, num_experts):
        """
        专家专业化程度：每个专家处理的token的特征相似度
        """
        batch_size, seq_len, feat_dim = token_features.shape
        features_flat = token_features.reshape(-1, feat_dim)
        assignments_flat = expert_assignments.reshape(-1)
        
        specialization_scores = []
        
        for expert_idx in range(num_experts):
            # 获取该专家处理的所有tokens
            expert_mask = (assignments_flat == expert_idx)
            
            if expert_mask.sum() < 2:
                specialization_scores.append(0.0)
                continue
            
            expert_features = features_flat[expert_mask]
            
            # 计算特征向量的平均余弦相似度
            features_norm = F.normalize(expert_features, dim=1)
            similarity_matrix = torch.mm(features_norm, features_norm.T)
            
            # 对角线元素设为0（自身相似度）
            mask = torch.eye(len(expert_features), device=similarity_matrix.device).bool()
            similarity_matrix.masked_fill_(mask, 0)
            
            # 平均相似度
            avg_similarity = similarity_matrix.sum() / (len(expert_features) * (len(expert_features) - 1))
            specialization_scores.append(avg_similarity.item())
        
        return torch.tensor(specialization_scores)
    
    @staticmethod
    def routing_consistency(expert_assignments, num_experts, k=2):
        """
        路由一致性：相似的token是否被路由到相同的专家
        """
        batch_size, seq_len = expert_assignments.shape
        
        # 计算每对token的路由相似度
        consistency_scores = []
        
        for i in range(batch_size):
            for j in range(i+1, batch_size):
                # 比较两个序列的路由决策
                assignments_i = expert_assignments[i]
                assignments_j = expert_assignments[j]
                
                # Jaccard相似度
                intersection = sum(1 for a, b in zip(assignments_i, assignments_j) if a == b)
                union = seq_len
                jaccard = intersection / union
                
                consistency_scores.append(jaccard)
        
        return torch.tensor(consistency_scores).mean().item() if consistency_scores else 0.0
```

```python
class MoEEfficiencyMetrics:
    """
    MoE计算效率评估
    """
    @staticmethod
    def flops_per_token(model, seq_len, batch_size=1):
        """
        计算每个token的FLOPs
        """
        total_flops = 0
        
        for name, module in model.named_modules():
            if 'moe' in name.lower() or 'expert' in name.lower():
                # MoE层的FLOPs计算
                if hasattr(module, 'num_experts') and hasattr(module, 'top_k'):
                    d_model = module.d_model
                    d_ff = getattr(module, 'd_ff', d_model * 4)
                    
                    # 门控网络FLOPs
                    gate_flops = 2 * d_model * module.num_experts * batch_size * seq_len
                    
                    # 专家网络FLOPs（实际激活的）
                    # 每个token激活top_k个专家
                    active_flops = (2 * d_model * d_ff + 2 * d_ff * d_model) * \
                                  module.top_k * batch_size * seq_len
                    
                    total_flops += gate_flops + active_flops
        
        flops_per_token = total_flops / (batch_size * seq_len)
        return flops_per_token
    
    @staticmethod
    def memory_usage(model, seq_len, batch_size=1):
        """
        估算MoE模型的内存使用
        """
        param_memory = 0
        activation_memory = 0
        
        for name, param in model.named_parameters():
            param_memory += param.numel() * param.element_size()
        
        # 估算激活内存
        for name, module in model.named_modules():
            if 'moe' in name.lower():
                if hasattr(module, 'num_experts'):
                    d_model = module.d_model
                    d_ff = getattr(module, 'd_ff', d_model * 4)
                    
                    # 门控激活
                    gate_activation = batch_size * seq_len * module.num_experts
                    
                    # 专家激活（只计算激活的专家）
                    expert_activation = batch_size * seq_len * module.top_k * d_ff
                    
                    activation_memory += (gate_activation + expert_activation) * 2  # 假设fp16
        
        return {
            'parameters_mb': param_memory / 1024 / 1024,
            'activations_mb': activation_memory / 1024 / 1024,
            'total_mb': (param_memory + activation_memory) / 1024 / 1024
        }
    
    @staticmethod
    def throughput_analysis(model, input_shape, device='cuda'):
        """
        吞吐量分析
        """
        import time
        
        model.eval()
        model.to(device)
        
        batch_size, seq_len, d_model = input_shape
        dummy_input = torch.randn(batch_size, seq_len, d_model).to(device)
        
        # 预热
        for _ in range(10):
            _ = model(dummy_input)
        
        # 测量
        torch.cuda.synchronize()
        start_time = time.time()
        
        num_iterations = 100
        for _ in range(num_iterations):
            _ = model(dummy_input)
        
        torch.cuda.synchronize()
        end_time = time.time()
        
        total_tokens = batch_size * seq_len * num_iterations
        throughput = total_tokens / (end_time - start_time)
        
        return {
            'throughput_tokens_per_sec': throughput,
            'latency_ms': (end_time - start_time) * 1000 / num_iterations,
            'batch_size': batch_size,
            'seq_len': seq_len
        }
```

```python
class MoEInferenceOptimizer:
    """
    MoE模型推理优化
    """
    def __init__(self, model, num_experts, top_k):
        self.model = model
        self.num_experts = num_experts
        self.top_k = top_k
        
        # 专家预测缓存
        self.expert_prediction_cache = {}
        
        # 动态专家选择
        self.expert_importance = torch.zeros(num_experts)
        self.expert_usage_count = torch.zeros(num_experts)
        
    def predict_expert_usage(self, input_features):
        """
        预测专家使用模式，提前加载专家权重
        """
        # 使用轻量级模型预测专家使用
        if not hasattr(self, 'expert_predictor'):
            self.expert_predictor = self._build_expert_predictor()
        
        # 预测哪些专家可能被使用
        predicted_experts = self.expert_predictor(input_features)
        
        # 预加载专家权重到GPU
        self._preload_experts(predicted_experts)
        
        return predicted_experts
    
    def _build_expert_predictor(self):
        """
        构建专家使用预测器
        """
        # 简单的基于历史使用的预测器
        class ExpertPredictor(nn.Module):
            def __init__(self, num_experts, input_dim=64):
                super().__init__()
                self.num_experts = num_experts
                self.predictor = nn.Sequential(
                    nn.Linear(input_dim, 128),
                    nn.ReLU(),
                    nn.Linear(128, num_experts),
                    nn.Sigmoid()
                )
            
            def forward(self, x):
                # x: [batch, seq, dim] -> 提取特征
                features = x.mean(dim=1)  # 平均池化
                scores = self.predictor(features)
                return scores > 0.5  # 二值预测
        
        return ExpertPredictor(self.num_experts)
    
    def _preload_experts(self, expert_mask):
        """
        预加载专家权重到GPU
        """
        for expert_idx in range(self.num_experts):
            if expert_mask[expert_idx]:
                # 确保专家在GPU上
                expert = self.model.get_expert(expert_idx)
                if not expert.weight.is_cuda:
                    expert.cuda()
            else:
                # 不常用的专家可以卸载到CPU
                expert = self.model.get_expert(expert_idx)
                if expert.weight.is_cuda and self.expert_usage_count[expert_idx] < 10:
                    expert.cpu()
    
    def dynamic_top_k(self, router_logits, current_load):
        """
        动态调整top-k值
        """
        # 根据当前负载调整k值
        avg_load = current_load.mean()
        
        if avg_load > 0.8:  # 高负载
            k = max(1, self.top_k - 1)  # 减少专家数量
        elif avg_load < 0.3:  # 低负载
            k = min(self.num_experts, self.top_k + 1)  # 增加专家数量
        else:
            k = self.top_k
        
        return k
```

```python
class MoECompressor:
    """
    MoE模型压缩
    """
    def __init__(self, model, compression_ratio=0.5):
        self.model = model
        self.compression_ratio = compression_ratio
        
    def prune_experts(self, importance_scores):
        """
        基于重要性剪枝专家
        """
        num_experts_to_keep = int(self.model.num_experts * self.compression_ratio)
        
        # 选择最重要的专家
        _, top_indices = torch.topk(importance_scores, k=num_experts_to_keep)
        
        # 创建新的MoE层
        new_moe = type(self.model)(
            d_model=self.model.d_model,
            d_ff=self.model.d_ff,
            num_experts=num_experts_to_keep,
            top_k=min(self.model.top_k, num_experts_to_keep)
        )
        
        # 复制重要的专家
        for new_idx, old_idx in enumerate(top_indices):
            new_moe.experts[new_idx].load_state_dict(
                self.model.experts[old_idx].state_dict()
            )
        
        # 调整门控网络
        new_moe.gate.weight.data = self.model.gate.weight.data[top_indices]
        if hasattr(self.model.gate, 'bias'):
            new_moe.gate.bias.data = self.model.gate.bias.data[top_indices]
        
        return new_moe
    
    def quantize_experts(self, bits=8):
        """
        专家量化
        """
        quantized_model = copy.deepcopy(self.model)
        
        for i, expert in enumerate(quantized_model.experts):
            for name, param in expert.named_parameters():
                if 'weight' in name:
                    # 动态量化
                    scale = param.abs().max() / (2**(bits-1) - 1)
                    quantized_param = torch.round(param / scale).clamp(
                        -2**(bits-1), 2**(bits-1)-1
                    )
                    setattr(expert, name, nn.Parameter(quantized_param * scale))
                    
                    # 保存量化参数
                    setattr(expert, f'{name}_scale', scale)
                    setattr(expert, f'{name}_zero_point', 0)
        
        return quantized_model
    
    def knowledge_distillation(self, teacher_model, student_model, 
                               temperature=2.0, alpha=0.5):
        """
        从大型MoE蒸馏到小型MoE
        """
        def distillation_loss(student_output, teacher_output, 
                             hard_labels, temperature, alpha):
            # 软目标损失
            soft_loss = nn.KLDivLoss(reduction='batchmean')(
                F.log_softmax(student_output / temperature, dim=-1),
                F.softmax(teacher_output / temperature, dim=-1)
            ) * (temperature ** 2)
            
            # 硬目标损失
            hard_loss = F.cross_entropy(student_output, hard_labels)
            
            # 结合损失
            return alpha * soft_loss + (1 - alpha) * hard_loss
        
        return distillation_loss
```

对于AI推理部署、编译优化、芯片开发，建议关注：
