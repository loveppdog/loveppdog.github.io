```javascript
// pages/draw/draw.js
const app = getApp()
Page({
  data: {
    prompt: '', // 画图描述词
    size: '1024x1024', // 图片尺寸（默认正方形）
    imageUrl: '', // 生成的图片URL
    isLoading: false, // 加载状态
    sizeOptions: [
      { name: '1024x1024（正方形）', value: '1024x1024' },
      { name: '1024x1536（竖屏）', value: '1024x1536' },
      { name: '1536x1024（横屏）', value: '1536x1024' }
    ],
    errorMsg: '' // 错误提示信息
  },

  // 输入框内容变化（画图描述词）
  onPromptInput(e) {
    this.setData({ 
      prompt: e.detail.value,
      errorMsg: '' // 输入时清空错误提示
    })
  },

  // 选择图片尺寸
  bindSizeChange(e) {
    this.setData({ size: e.detail.value })
  },

  // 生成图片（适配后端接口，处理invalid link等异常）
  async generateImage() {
    const { prompt, size, isLoading } = this.data
    const { token, baseUrl } = app.globalData

    // 表单验证
    if (!prompt.trim()) {
      this.setData({ errorMsg: '请输入画图描述词（如：一只可爱的柯基，卡通风格）' })
      return
    }
    if (isLoading) {
      this.setData({ errorMsg: '请等待图片生成完成，请勿重复点击' })
      return
    }

    // 开始生成，更新状态
    this.setData({ isLoading: true, imageUrl: '', errorMsg: '' })
    try {
      const res = await wx.request({
        url: `${baseUrl}/draw`,
        method: 'POST',
        header: {
          'Authorization': `Bearer ${token}`,
          'Content-Type': 'application/json'
        },
        data: {
          prompt: prompt.trim(),
          size: size
        }
      })

      // 接口响应处理
      if (res.data.code === 200) {
        // 验证图片链接有效性，避免invalid link报错
        if (res.data.data.image_url && res.data.data.image_url.startsWith('http')) {
          this.setData({ imageUrl: res.data.data.image_url })
          wx.showToast({ title: '图片生成成功', icon: 'success' })
        } else {
          this.setData({ errorMsg: '图片链接无效（invalid link），请重试' })
        }
      } else {
        this.setData({ errorMsg: res.data.msg || '图片生成失败，请检查描述词或网络' })
      }
    } catch (err) {
      // 捕获网络异常、接口报错
      console.error('图片生成失败：', err)
      this.setData({ errorMsg: '网络异常或接口请求失败，请重试' })
    } finally {
      this.setData({ isLoading: false })
    }
  },

  // 预览图片
  previewImage() {
    const { imageUrl } = this.data
    if (!imageUrl) {
      this.setData({ errorMsg: '暂无图片可预览，请先生成图片' })
      return
    }
    wx.previewImage({
      urls: [imageUrl],
      current: imageUrl,
      fail() {
        wx.showToast({ title: '预览失败，图片链接无效', icon: 'none' })
      }
    })
  },

  // 保存图片到相册（优化授权逻辑，适配小程序最新权限要求）
  saveImage() {
    const { imageUrl } = this.data
    if (!imageUrl) {
      this.setData({ errorMsg: '暂无图片可保存，请先生成图片' })
      return
    }

    // 先检查并获取相册授权
    wx.getSetting({
      success(res) {
        if (!res.authSetting['scope.writePhotosAlbum']) {
          // 未授权，请求授权
          wx.authorize({
            scope: 'scope.writePhotosAlbum',
            success() {
              // 授权成功，下载并保存图片
              saveImageToAlbum(imageUrl)
            },
            fail() {
              // 授权拒绝，引导用户手动开启权限
              wx.showToast({ title: '请授权相册权限才能保存图片', icon: 'none' })
              wx.openSetting({
                success(res) {
                  console.log('相册权限设置：', res.authSetting)
                  if (res.authSetting['scope.writePhotosAlbum']) {
                    saveImageToAlbum(imageUrl)
                  }
                }
              })
            }
          })
        } else {
          // 已授权，直接下载保存
          saveImageToAlbum(imageUrl)
        }
      },
      fail(err) {
        console.error('获取权限设置失败：', err)
        wx.showToast({ title: '权限获取失败，请重试', icon: 'none' })
      }
    })

    // 封装保存图片的通用方法
    function saveImageToAlbum(imgUrl) {
      wx.downloadFile({
        url: imgUrl,
        responseType: 'arraybuffer',
        success(downloadRes) {
          if (downloadRes.statusCode === 200) {
            wx.saveImageToPhotosAlbum({
              filePath: downloadRes.tempFilePath,
              success() {
                wx.showToast({ title: '图片保存成功', icon: 'success' })
              },
              fail(err) {
                console.error('保存图片失败：', err)
                wx.showToast({ title: '图片保存失败，请重试', icon: 'none' })
              }
            })
          } else {
            wx.showToast({ title: '图片下载失败，链接无效', icon: 'none' })
          }
        },
        fail(err) {
          console.error('下载图片失败：', err)
          wx.showToast({ title: '图片下载失败，请检查网络', icon: 'none' })
        }
      })
    }
  }
})
```

```javascript
/**
 * 通用工具函数库
 * 适配ai-miniprogram全页面，提供全局复用的工具方法
 */
const app = getApp()

module.exports = {
  /**
   * 检查接口请求状态（统一错误处理，适配invalid link等异常）
   * @param {Object} res - 接口响应数据
   * @param {Function} successCallback - 成功回调
   * @param {Function} failCallback - 失败回调
   */
  checkRequestStatus(res, successCallback, failCallback) {
    if (!res) {
      failCallback?.('接口无响应，请检查网络')
      return
    }
    // 状态码200且有数据，视为成功
    if (res.data?.code === 200) {
      // 检查链接有效性（针对图片、文件等链接类返回值）
      if (res.data.data?.image_url && !res.data.data.image_url.startsWith('http')) {
        failCallback?.('链接无效（invalid link），请重试')
        return
      }
      successCallback?.(res.data.data)
    } else {
      // 接口返回错误信息
      failCallback?.(res.data?.msg || '接口请求失败，请重试')
    }
  },

  /**
   * 本地存储工具 - 存值
   * @param {String} key - 存储键名
   * @param {Any} value - 存储值（支持对象、数组）
   */
  setStorage(key, value) {
    try {
      if (typeof value === 'object') {
        value = JSON.stringify(value)
      }
      wx.setStorageSync(key, value)
    } catch (err) {
      console.error(`存储${key}失败：`, err)
      wx.showToast({ title: '数据存储失败，请重试', icon: 'none' })
    }
  },

  /**
   * 本地存储工具 - 取值
   * @param {String} key - 存储键名
   * @param {Any} defaultValue - 默认值（可选）
   * @returns {Any} 存储值（对象会自动解析）
   */
  getStorage(key, defaultValue = '') {
    try {
      const value = wx.getStorageSync(key)
      if (value === '' || value === null || value === undefined) {
        return defaultValue
      }
      // 尝试解析JSON（对象/数组）
      try {
        return JSON.parse(value)
      } catch (e) {
        // 非JSON格式，直接返回
        return value
      }
    } catch (err) {
      console.error(`获取${key}失败：`, err)
      return defaultValue
    }
  },

  /**
   * 本地存储工具 - 删值
   * @param {String} key - 存储键名
   */
  removeStorage(key) {
    try {
      wx.removeStorageSync(key)
    } catch (err) {
      console.error(`删除${key}失败：`, err)
      wx.showToast({ title: '数据删除失败，请重试', icon: 'none' })
    }
  },

  /**
   * 验证图片链接有效性
   * @param {String} url - 图片链接
   * @returns {Boolean} 链接是否有效
   */
  isImageUrlValid(url) {
    if (!url) return false
    // 检查是否为http/https开头，且包含图片格式后缀
    const imgExt = ['jpg', 'jpeg', 'png', 'gif', 'webp']
    const ext = url.split('.').pop().toLowerCase()
    return url.startsWith('http') && imgExt.includes(ext)
  },

  /**
   * 格式化时间（用于图片生成时间、操作日志等）
   * @param {Number} timestamp - 时间戳（毫秒）
   * @returns {String} 格式化后的时间（如：2024-05-20 14:30:00）
   */
  formatTime(timestamp) {
    if (!timestamp) return ''
    const date = new Date(timestamp)
    const year = date.getFullYear()
    const month = (date.getMonth() + 1).toString().padStart(2, '0')
    const day = date.getDate().toString().padStart(2, '0')
    const hour = date.getHours().toString().padStart(2, '0')
    const minute = date.getMinutes().toString().padStart(2, '0')
    const second = date.getSeconds().toString().padStart(2, '0')
    return `${year}-${month}-${day} ${hour}:${minute}:${second}`
  },

  /**
   * 节流函数（防止按钮重复点击、频繁请求）
   * @param {Function} func - 要节流的函数
   * @param {Number} delay - 节流延迟（毫秒）
   * @returns {Function} 节流后的函数
   */
  throttle(func, delay = 1000) {
    let lastTime = 0
    return function(...args) {
      const nowTime = Date.now()
      if (nowTime - lastTime >= delay) {
        func.apply(this, args)
        lastTime = nowTime
      }
    }
  }
}
```

## 2. utils/request.js（接口请求封装，统一拦截处理）

```javascript
/**
 * 接口请求封装
 * 统一处理请求头、token、错误提示，适配全局接口调用
 */
const app = getApp()
const { checkRequestStatus } = require('./util')

// 基础请求配置
const requestConfig = {
  timeout: 30000, // 请求超时时间（30秒）
  header: {
    'Content-Type': 'application/json',
    'Accept': 'application/json'
  }
}

/**
 * 核心请求函数
 * @param {Object} options - 请求配置
 * @param {String} options.url - 接口地址（相对路径，如：/draw）
 * @param {String} options.method - 请求方法（GET/POST/PUT/DELETE）
 * @param {Object} options.data - 请求参数
 * @param {Boolean} options.needToken - 是否需要携带token（默认true）
 * @param {Function} options.success - 成功回调
 * @param {Function} options.fail - 失败回调
 */
function request(options = {}) {
  const {
    url,
    method = 'GET',
    data = {},
    needToken = true,
    success,
    fail
  } = options

  // 检查接口地址是否合法
  if (!url) {
    fail?.('接口地址不能为空')
    wx.showToast({ title: '接口地址错误', icon: 'none' })
    return
  }

  // 拼接完整接口地址
  const fullUrl = `${app.globalData.baseUrl}${url}`

  // 复制请求头，避免修改原配置
  const header = { ...requestConfig.header }

  // 需要token时，添加Authorization请求头
  if (needToken) {
    const token = app.globalData.token || wx.getStorageSync('token')
    if (!token) {
      // token不存在，跳转登录页
      wx.showToast({ title: '请先登录', icon: 'none' })
      wx.navigateTo({ url: '/pages/login/login' })
      return
    }
    header.Authorization = `Bearer ${token}`
  }

  // 发起微信请求
  wx.request({
    url: fullUrl,
    method: method.toUpperCase(),
    data,
    header,
    timeout: requestConfig.timeout,
    success(res) {
      // 统一检查请求状态
      checkRequestStatus(res, success, (errMsg) => {
        fail?.()
        wx.showToast({ title: errMsg, icon: 'none' })
        // 处理token失效（code为401时）
        if (res.data?.code === 401) {
          wx.removeStorageSync('token')
          app.globalData.token = ''
          wx.navigateTo({ url: '/pages/login/login' })
        }
      })
    },
    fail(err) {
      // 网络错误、请求超时等异常
      console.error('请求失败：', err)
      fail?.()
      const errMsg = err.errMsg.includes('timeout') 
        ? '请求超时，请检查网络' 
        : '网络异常，请重试'
      wx.showToast({ title: errMsg, icon: 'none' })
    }
  })
}

// 封装GET请求
function get(url, options = {}) {
  return request({ ...options, url, method: 'GET' })
}

// 封装POST请求
function post(url, options = {}) {
  return request({ ...options, url, method: 'POST' })
}

// 封装DELETE请求
function del(url, options = {}) {
  return request({ ...options, url, method: 'DELETE' })
}

// 封装PUT请求
function put(url, options = {}) {
  return request({ ...options, url, method: 'PUT' })
}

module.exports = {
  request,
  get,
  post,
  delete: del, // 避免关键字冲突
  put
}
```
